// Audio Recorder Engine (Fix Format & MimeType) let mediaRecorder, audioChunks = [], audioBase64 = null; async function toggleRecording() { const btn = document.getElementById('recBtn'); if (!mediaRecorder || mediaRecorder.state === "inactive") { try { const stream = await navigator.mediaDevices.getUserMedia({ audio: true }); // Pilih MIME type yang didukung oleh browser (termasuk Safari/Chrome) let mimeType = 'audio/webm'; if (!MediaRecorder.isTypeSupported(mimeType)) { if (MediaRecorder.isTypeSupported('audio/mp4')) { mimeType = 'audio/mp4'; } else if (MediaRecorder.isTypeSupported('audio/ogg')) { mimeType = 'audio/ogg'; } else { mimeType = ''; // Menggunakan default browser } } mediaRecorder = mimeType ? new MediaRecorder(stream, { mimeType }) : new MediaRecorder(stream); audioChunks = []; mediaRecorder.ondataavailable = e => { if (e.data.size > 0) audioChunks.push(e.data); }; mediaRecorder.onstop = async () => { const actualType = mediaRecorder.mimeType || 'audio/webm'; const blob = new Blob(audioChunks, { type: actualType }); // Set sumber audio preview const audioPreview = document.getElementById('audioPreview'); audioPreview.src = URL.createObjectURL(blob); audioPreview.classList.remove('hidden'); audioPreview.load(); // Refresh elemen audio agar tidak Error // Convert ke Base64 const reader = new FileReader(); reader.readAsDataURL(blob); reader.onloadend = () => { audioBase64 = reader.result; }; }; mediaRecorder.start(); btn.innerText = "⏹️ Hentikan Rekaman"; btn.classList.replace('bg-rose-500', 'bg-slate-700'); } catch (err) { alert("Izin akses mikrofon ditolak atau tidak didukung di perangkat ini."); } } else { mediaRecorder.stop(); btn.innerText = "🎙️ Rekam Ulang"; btn.classList.replace('bg-slate-700', 'bg-rose-500'); } }
